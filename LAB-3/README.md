def swap(arr, i, j):
    """Міняє місцями два елементи в масиві."""
    arr[i], arr[j] = arr[j], arr[i]

def sink(arr, i, n):
    """
    Процедура 'занурення' елемента вниз по купі з трасуванням.
    arr: масив
    i: індекс поточного елемента
    n: поточний розмір купи
    """
    k = i
    while True:
        j = 2 * k + 1  # Індекс лівого дочірнього елемента
        if j >= n:
            print(f"  Елемент arr[{k}]={arr[k]} на індексі {k} досяг кінця купи. Завершуємо.")
            break
        
        # Знаходимо найбільший з дочірніх елементів
        if j + 1 < n and arr[j + 1] > arr[j]:
            print(f"  Порівнюємо лівий arr[{j}]={arr[j]} та правий arr[{j+1}]={arr[j+1]}. Обираємо правий на індексі {j+1}.")
            j += 1
        else:
            print(f"  Обираємо лівий дочірній елемент arr[{j}]={arr[j]} на індексі {j}.")
            
        # Перевіряємо умову купи
        if arr[k] >= arr[j]:
            print(f"  arr[{k}]={arr[k]} (батько) >= arr[{j}]={arr[j]} (дочірній). Елемент на своєму місці.")
            break
        else:
            print(f"  Міняємо місцями arr[{k}]={arr[k]} (батько) та arr[{j}]={arr[j]} (дочірній).")
            swap(arr, k, j)
            print(f"  Масив після обміну: {arr}")
            k = j

def heapsort(arr):
    n = len(arr)
    print(f"Початковий масив: {arr}\n")
    
    # Фаза 1: Побудова максимальної купи
    print("--- Фаза 1: Побудова максимальної купи ---")
    start_idx = (n // 2) - 1
    for i in range(start_idx, -1, -1):
        print(f"\nПочинаємо 'занурювати' елемент arr[{i}]={arr[i]} з індексу {i}:")
        sink(arr, i, n)
        
    print(f"\nМасив після побудови купи: {arr}\n")
    
    # Фаза 2: Сортування
    print("--- Фаза 2: Сортування ---")
    for i in range(n - 1, 0, -1):
        print(f"\nМіняємо місцями корінь (arr[0]={arr[0]}) та останній елемент купи (arr[{i}]={arr[i]})")
        swap(arr, 0, i)
        print(f"Розмір купи зменшився до {i}. Відновлюємо властивості купи.")
        sink(arr, 0, i)
        print(f"Масив на поточному кроці: {arr}")
        
    return arr

A = [69, 52, 97, 27, 10, 88, 29, 1, 24]
sorted_A = heapsort(A)
print(f"\nВідсортований масив: {sorted_A}")
